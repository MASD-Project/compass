/**
 * @brief Whether a claim is checked against the row its key names now.
 */
bool run_claim_reads_current(const ores::utility::domain::precondition& claim) {
    using ores::utility::domain::precondition_kind;
    // No version to state, so only a create needs to know whether a row exists.
    return claim.kind == precondition_kind::must_not_exist;
}

using run_rows_by_key = std::map<std::tuple<std::string>, std::vector<domain::run>>;

/**
 * @brief The current rows the objects name, by the text form of their key.
 *
 * One batch read for the set, so a batch checks its claims against one read
 * rather than one read per row.
 */
run_rows_by_key run_current_rows(run_repository& repo,
                                 const ores::database::context& ctx,
                                 const std::vector<domain::run>& of) {
    std::vector<std::string> ids;
    ids.reserve(of.size());
    for (const auto& row : of) {
        ids.push_back(std::to_string(row.id));
    }
    run_rows_by_key current;
    for (auto& item : repo.read_latest(ctx, ids))
        current[std::make_tuple(std::to_string(item.id))].push_back(std::move(item));
    return current;
}

/**
 * @brief The current rows that the key of an object names, or none.
 */
const std::vector<domain::run>& run_rows_of(const run_rows_by_key& current,
                                            const domain::run& item) {
    static const std::vector<domain::run> none;
    const auto found = current.find(std::make_tuple(std::to_string(item.id)));
    return found == current.end() ? none : found->second;
}

/**
 * @brief The claim a replace makes: the version the row carries now, or
 * that no row exists yet.
 */
ores::utility::domain::precondition run_replace_claim(const std::vector<domain::run>& current) {
    if (current.empty())
        return {ores::utility::domain::precondition_kind::must_not_exist, std::nullopt};
    // No version column to state, so a replace names no claim at all; the
    // store replaces the row as it stands. A create over a live row is refused
    // when the claim is applied.
    return {ores::utility::domain::precondition_kind::any, std::nullopt};
}

/**
 * @brief The object with the claim's version stamped onto it.
 *
 * The caller passes the row the key names now whenever the claim reads it. A
 * claim the store cannot check is refused here rather than ignored.
 */
domain::run run_apply_claim(const domain::run& v,
                            const ores::utility::domain::precondition& claim,
                            const std::vector<domain::run>& current) {
    using ores::utility::domain::precondition_kind;
    auto t = v;
    // No version column to state, so the claim is honoured by the read alone.
    if (claim.kind == precondition_kind::must_match_version)
        throw std::invalid_argument("run_repository::write: this table keeps no version to match");
    if (claim.kind == precondition_kind::must_not_exist && !current.empty())
        throw std::invalid_argument("run_repository::write: a current row already exists");
    return t;
}

}

void run_repository::write(const context& ctx, const domain::run& v) {
    write(ctx, v, run_replace_claim(read_latest(ctx, std::to_string(v.id))));
}

void run_repository::write(const context& ctx, const std::vector<domain::run>& v) {
    const auto current = run_current_rows(*this, ctx, v);
    std::vector<ores::utility::domain::precondition> claims;
    claims.reserve(v.size());
    for (const auto& item : v)
        claims.push_back(run_replace_claim(run_rows_of(current, item)));
    write(ctx, v, claims);
}

void run_repository::write(const context& ctx,
                           const domain::run& v,
                           const ores::utility::domain::precondition& claim) {
    ORES_LOG_SEV(lg(), debug) << "Writing run. " << "id: " << v.id;
    const auto t =
        run_apply_claim(v,
                        claim,
                        run_claim_reads_current(claim) ? read_latest(ctx, std::to_string(v.id)) :
                                                         std::vector<domain::run>{});
    query_builder::execute_write_op(ctx,
                                    query_builder::insert_or_replace(run_mapper::map(t)),
                                    lg(),
                                    "Writing run to database.");
}

void run_repository::write(const context& ctx,
                           const std::vector<domain::run>& v,
                           const std::vector<ores::utility::domain::precondition>& claims) {
    ORES_LOG_SEV(lg(), debug) << "Writing runs. Count: " << v.size();
    if (claims.size() != v.size())
        throw std::invalid_argument("run_repository::write: one claim is needed for each row");
    std::vector<domain::run> to_read;
    for (std::size_t i = 0; i < v.size(); ++i) {
        if (run_claim_reads_current(claims[i]))
            to_read.push_back(v[i]);
    }
    const auto current = run_current_rows(*this, ctx, to_read);
    std::vector<domain::run> batch;
    batch.reserve(v.size());
    for (std::size_t i = 0; i < v.size(); ++i)
        batch.push_back(run_apply_claim(v[i], claims[i], run_rows_of(current, v[i])));
    query_builder::execute_write_op(ctx,
                                    query_builder::insert_or_replace(run_mapper::map(batch)),
                                    lg(),
                                    "Writing runs to database.");
}

std::vector<domain::run> run_repository::read_latest(const context& ctx) {
    const auto query =
        query_builder::read<std::vector<run_entity>> | query_builder::order_by("id"_c);

    return query_builder::execute_read_query<run_entity, domain::run>(

std::vector<domain::run> run_repository::read_latest(const context& ctx,
                                                     const std::vector<std::string>& ids) {
    if (ids.empty())
        return {};
    // Every key is a bind and the store's bind count is finite, so a larger
    // key set is read one chunk at a time.
    constexpr std::size_t chunk_size = 1000;
    if (ids.size() > chunk_size) {
        std::vector<domain::run> all;
        for (std::size_t begin = 0; begin < ids.size(); begin += chunk_size) {
            const auto end = std::min(begin + chunk_size, ids.size());
            auto part =
                read_latest(ctx, std::vector<std::string>(ids.begin() + begin, ids.begin() + end));
            all.insert(all.end(),
                       std::make_move_iterator(part.begin()),
                       std::make_move_iterator(part.end()));
        }
        return all;
    }
    const auto query =
        query_builder::read<std::vector<run_entity>> | query_builder::where("id"_c.in(ids));
    auto result = query_builder::execute_read_query<run_entity, domain::run>(
        ctx,
        query,
        [](const auto& entities) { return run_mapper::map(entities); },
        lg(),
        "Reading latest runs by ids.");
    return result;
}

void run_repository::remove(const context& ctx, const std::vector<std::string>& ids) {
    // A batch of nothing addresses no row, so there is nothing to delete. The
    // query builder renders an empty key list as an empty IN (), which the
    // server refuses as a syntax error; the read overloads answer the empty
    // case the same way. The compound branch above is left alone: it loops, so
    // it already removes nothing, and its length check still refuses an
    // asymmetric pair.
    if (ids.empty())
        return;
    const auto query =
        query_builder::delete_from<run_entity> | query_builder::where("id"_c.in(ids));
    query_builder::execute_delete_query(ctx, query, lg(), "Batch removing runs.");
}

/**
 * @brief The current row a key names, or an empty vector when there is none.
 *
 * A key record carries each column with the column's own type, and the
 * repository takes the text form every one of its key parameters shares, so
 * the conversion lives here rather than at every call site.
 *
 * The key record carries the key the model declares, which is the one a caller
 * holds. When that is not the storage key the row is found by it and the
 * repository's storage-key read is not used at all.
 */
std::vector<domain::run> run_read_one(repository::run_repository& repo,
                                      const ores::database::context& ctx,
                                      const messaging::run_key& key) {
    return repo.read_latest(ctx, std::to_string(key.id));
}

/**
 * @brief The current rows a key set names, one entry per key in the keys' order.
 *
 * Each entry holds what the one-key read holds for its key, so a batch checks
 * each row against the same answer a single request gets. The repository
 * reads the key set in one batch read.
 *
 * A row is joined back to its key by the text form of each key column. Both
 * sides convert the typed value with the same function, and the store compares
 * key text exactly, so every row the read returns finds its key.
 *
 * When the declared key is not the storage key the batch is read by the
 * declared key instead. A model that states its own read by that key has no
 * batch form of it, so its keys are read one at a time.
 */
std::vector<std::vector<domain::run>> run_read_many(repository::run_repository& repo,
                                                    const ores::database::context& ctx,
                                                    const std::vector<messaging::run_key>& keys) {
    std::vector<std::vector<domain::run>> found(keys.size());
    using key_tuple = std::tuple<std::string>;
    std::vector<std::string> id_keys;
    id_keys.reserve(keys.size());
    for (const auto& key : keys) {
        id_keys.push_back(std::to_string(key.id));
    }
    std::map<key_tuple, std::vector<domain::run>> by_key;
    for (auto& row : repo.read_latest(ctx, id_keys)) {
        key_tuple named(std::to_string(row.id));
        by_key[std::move(named)].push_back(std::move(row));
    }
    for (std::size_t i = 0; i < keys.size(); ++i) {
        const auto& key = keys[i];
        const auto rows = by_key.find(key_tuple(std::to_string(key.id)));
        if (rows != by_key.end())
            found[i] = rows->second;
    }
    return found;
}

messaging::list_runs_response run_service::list_runs(const messaging::list_runs_request& request) {
    messaging::list_runs_response response;
    if (!request.order.field.empty() &&
        !repository::run_repository::is_sortable(request.order.field)) {
        response.result = refuse(outcome_code::order_not_supported,
                                 {.entity = "runs", .field = request.order.field});
        return response;
    }
    if (request.filter && request.filter->id_one_of &&
        request.filter->id_one_of->size() > max_batch_size) {
        response.result = refuse(outcome_code::filter_too_large,
                                 {.field = "id_one_of", .limit = std::to_string(max_batch_size)});
        return response;
    }
    response.=
        repo_.read_latest(ctx_, request.offset, request.limit, request.order, request.filter);
    response.total = repo_.get_total__count(ctx_, request.filter);
    return response;
}

messaging::get_run_response run_service::get_run(const messaging::get_run_request& request) {
    messaging::get_run_response response;
    auto found = run_read_one(repo_, ctx_, request.key);
    if (found.empty()) {
        response.result = refuse(outcome_code::not_found, {.entity = "run"});
        return response;
    }
    response.run = std::move(found.front());
    return response;
}

messaging::get_many_runs_response
run_service::get_many_runs(const messaging::get_many_runs_request& request) {
    messaging::get_many_runs_response response;
    // A key set carries at most as many keys as the protocol lets a batch
    // carry, so one request cannot ask for an unbounded reply.
    if (request.keys.size() > max_batch_size) {
        response.result = refuse(outcome_code::filter_too_large,
                                 {.field = "keys", .limit = std::to_string(max_batch_size)});
        return response;
    }
    // One entry per requested key, in the order asked for, so the reply is
    // positional and a caller reads absence from an empty entry rather than
    // from a missing one.
    response.entries.reserve(request.keys.size());
    auto found = run_read_many(repo_, ctx_, request.keys);
    for (std::size_t i = 0; i < request.keys.size(); ++i) {
        messaging::run_lookup entry;
        entry.key = request.keys[i];
        if (!found[i].empty())
            entry.run = std::move(found[i].front());
        response.entries.push_back(std::move(entry));
    }
    return response;
}
